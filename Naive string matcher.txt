#Naive string matcher algorithm
n = input("Enter the number of characters in Text: ")
n = int(n)
while n <= 0:
	print("Enter the number of characters must be greater than zero.")
	n = input("Enter the number of characters in Text: ")
	n = int(n)
mytext = []
for i in range(1,n+1):
    character = input("Enter the character number "+str(i)+":  ")
    mytext.append(character)

print("Your text is: "+str(mytext)+".")

m = input("Enter the number of characters in Pattern: ")
m = int(m)

while (m > n or m <= 0):
	print("The length of the pattern must be less than or equal to the length of the text and greater than zero.")
	m = input("Enter the number of characters in Pattern: ")
	m = int(m)

pattern = []
for i in range(1,m+1):
	character = input("Enter the character number "+str(i)+":  ")
	pattern.append(character)
		
print("Your pattern is: "+str(pattern)+".")

for s in range(0,n-m+1):
	j=0
	while (j <= m-1 and mytext[s+j] == pattern[j]):
		j +=1
	if j > m-1:
		print(str(s)+" is a valid shift.")
